
1. INPUT N
2. IF N <= 2 OR N >= 10 THEN
    IF N <= 2 THEN
        OUTPUT "Terlalu Kecil"
    ELSE
        OUTPUT "Terlalu Besar"
    ENDIF
    RETURN
ENDIF

3. ANSWER = ""
4. FOR I = 0 TO N-1 DO
    FOR J = 0 TO N-1 DO
        IF I = 0 OR I = N-1 THEN
            OUTPUT "="
            ANSWER <- ANSWER + "="
        ELSE
            IF J = 0 OR J = N-1 THEN
                OUTPUT "|"
                ANSWER <- ANSWER + "|"
            ELSE
                OUTPUT " "
                ANSWER <- ANSWER + " "
            ENDIF
        ENDIF
    ENDFOR
    OUTPUT ""
ENDFOR
OUTPUT ANSWER
